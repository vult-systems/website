{/*
  Bento grid sized from each image's real aspect ratio (no preset buckets,
  no cropping). Row/gap/column units are all expressed in cqw (% of the grid's
  own container width), not px, so the exact aspect ratio holds true at every
  viewport instead of only the one width it was tuned against.
*/}
export const GAP_CQW = 1.8;
export const ROW_CQW = 0.9;
export const COL_CQW = (100 - GAP_CQW) / 2;
// Row-span count needed for an image rendered at a given % of the grid's
// own width (cqw units), so this always matches the item's ACTUAL rendered
// width rather than assuming a fixed column count.
export const rowSpanForWidth = (img, widthCqw) => {
  const height = (img.height / img.width) * widthCqw;
  return Math.max(1, Math.ceil((height + GAP_CQW) / (ROW_CQW + GAP_CQW)));
};
// Desktop (sm:grid-cols-2): non-wide items sit in one of two columns
// (COL_CQW wide), wide items span both (100 wide).
export const rowSpan = (img, cols = 1) => rowSpanForWidth(img, cols === 2 ? 100 : COL_CQW);
// "small" items stay a completely normal single-column grid item (same
// col-span, same dense-packing behavior as everything else) -- they just
// render at a reduced width/height *within* that same reserved cell,
// computed from the SAME reduced width used for both dimensions, so the
// card's own aspect ratio still exactly matches the image's real ratio and
// there's no dead space inside the border. The only visible cost is a
// strip of ordinary page background beside/below it in its own cell,
// which is fine -- unlike giving it a genuinely narrower column-span, nothing
// downstream in the grid depends on that space being reclaimed, so this
// can never produce the dense-packing gaps a partial-track "wide"-style
// item can (see git history for two rounds of exactly that bug).
export const SMALL_SCALE = 0.6;

<style>{`
  /* grid-cols-1 below sm means every item (wide or not) actually renders at
     100% of the grid's width, not the desktop column width rowSpan() below
     assumes — so row height needs its own mobile-width calculation instead
     of reusing the desktop one, or short items end up reserved half the
     height they need and float tiny inside a too-short cell. */
  .sketchbook-item { grid-row: span var(--row-mobile); }
  @media (min-width: 640px) {
    .sketchbook-item { grid-row: span var(--row-desktop); }
  }
  /* "small" items render at a reduced width within their normal grid cell
     (see SMALL_SCALE above) instead of stretching to fill it -- the row
     height above is computed from that same reduced width, so the card's
     own box still matches its image's real aspect ratio exactly. */
  .small-item { width: var(--item-width); }
`}</style>
<div class="my-8 [container-type:inline-size]">
  <div class="grid grid-cols-1 sm:grid-cols-2 [grid-auto-flow:dense]" style={`gap: ${GAP_CQW}cqw; grid-auto-rows: ${ROW_CQW}cqw;`}>
    {sketches.map(({ img, alt, wide, small }) => {
      const scale = small ? SMALL_SCALE : 1;
      const desktopWidthCqw = (wide ? 100 : COL_CQW) * scale;
      const mobileWidthCqw = 100 * scale;
      return (
        <div
          class={`sketchbook-item spotlight-card group relative ${wide ? 'sm:col-span-2' : ''} ${small ? 'small-item' : ''}`}
          style={`--row-mobile: ${rowSpanForWidth(img, mobileWidthCqw)}; --row-desktop: ${rowSpanForWidth(img, desktopWidthCqw)}; --item-width: ${scale * 100}%;`}
        >
          <div class="spotlight-inner h-full rounded-md overflow-hidden bg-card transition-all duration-300 group-hover:scale-[1.02]">
            <img src={img.src} alt={alt} class="w-full h-full object-contain !my-0" />
          </div>
        </div>
      );
    })}
  </div>
</div>
