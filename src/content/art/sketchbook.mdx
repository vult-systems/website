{/*
  Bento grid sized from each image's real aspect ratio (no preset buckets,
  no cropping). Row/gap/column units are all expressed in cqw (% of the grid's
  own container width), not px, so the exact aspect ratio holds true at every
  viewport instead of only the one width it was tuned against.
*/}
export const GAP_CQW = 1.8;
export const ROW_CQW = 0.9;
export const COL_CQW = (100 - GAP_CQW) / 2;
// Row-span count needed for an image rendered at a given % of the grid's
// own width (cqw units), so this always matches the item's ACTUAL rendered
// width rather than assuming a fixed column count.
export const rowSpanForWidth = (img, widthCqw) => {
  const height = (img.height / img.width) * widthCqw;
  return Math.max(1, Math.ceil((height + GAP_CQW) / (ROW_CQW + GAP_CQW)));
};
// Desktop (sm:grid-cols-2): non-wide items sit in one of two columns
// (COL_CQW wide), wide items span both (100 wide).
export const rowSpan = (img, cols = 1) => rowSpanForWidth(img, cols === 2 ? 100 : COL_CQW);

     height they need and float tiny inside a too-short cell. */
  .sketchbook-item { grid-row: span var(--row-mobile); }
  @media (min-width: 640px) {
    .sketchbook-item { grid-row: span var(--row-desktop); }
  }
`}</style>
